#### RoomHandler - 房间事件处理
```typescript
export class RoomHandler {
  private roomService: RoomService
  private io: Server

  constructor(roomService: RoomService, io: Server) {
    this.roomService = roomService
    this.io = io
  }

  registerHandlers(socket: Socket): void {
    // 创建房间
    socket.on('room:create', (data) => this.handleCreateRoom(socket, data))
    
    // 加入房间
    socket.on('room:join', (data) => this.handleJoinRoom(socket, data))
    
    // 离开房间
    socket.on('room:leave', (data) => this.handleLeaveRoom(socket, data))
    
    // 准备/取消准备
    socket.on('room:ready', (data) => this.handleReady(socket, data))
    
    // 断开连接
    socket.on('disconnect', () => this.handleDisconnect(socket))
  }

  private handleCreateRoom(socket: Socket, data: { name: string; maxPlayers: number }): void
  private handleJoinRoom(socket: Socket, data: { roomId: string; nickname: string }): void
  private handleLeaveRoom(socket: Socket, data: { roomId: string }): void
  private handleReady(socket: Socket, data: { roomId: string }): void
  private handleDisconnect(socket: Socket): void
}
```

## 五、游戏流程设计

### 5.1 游戏状态机
```
┌──────────────────────────────────────────────────────────────┐
│                      游戏流程状态机                           │
├──────────────────────────────────────────────────────────────┤
│                                                              │
│   ┌─────────┐    所有玩家准备     ┌─────────────┐            │
│   │  等待   │ ─────────────────→ │  选词阶段   │            │
│   │ WAITING │                    │  SELECTING  │            │
│   └─────────┘                    └──────┬──────┘            │
│        ↑                                │                    │
│        │                          选词完成│                    │
│        │                          (10秒)  │                    │
│        │                                ↓                    │
│   ┌────┴────┐    时间到/所有人猜对 ┌─────────────┐            │
│   │ 结算    │ ←────────────────── │  绘画阶段   │            │
│   │ROUND_END│                     │  DRAWING   │            │
│   └────┬────┘                     └─────────────┘            │
│        │                                                       │
│        │ 显示答案(5秒)                                          │
│        ↓                                                       │
│   ┌─────────┐                                                  │
│   │ 下一轮  │ ──→ 循环或结束                                    │
│   │NEXT_ROUND│                                                 │
│   └─────────┘                                                  │
│        │                                                       │
│        │ 所有回合结束                                           │
│        ↓                                                       │
│   ┌─────────┐                                                  │
│   │ 游戏结束│                                                  │
│   │GAME_END │                                                  │
│   └─────────┘                                                  │
│                                                               │
└──────────────────────────────────────────────────────────────┘
```

### 5.2 详细流程说明

#### 阶段1: 等待阶段 (WAITING)
- 玩家加入房间
- 玩家点击准备
- 房主点击开始游戏
- **条件**: 至少2名玩家，所有玩家已准备

#### 阶段2: 选词阶段 (SELECTING)
- 服务端随机选择3个词语
- 发送给当前画手
- 画手选择一个词语
- **超时**: 10秒自动选择第一个

#### 阶段3: 绘画阶段 (DRAWING)
- 画手在画布上绘画
- 其他玩家猜词
- 服务端广播画布数据
- 服务端广播倒计时
- **结束条件**: 时间到(60秒) 或 所有玩家猜对

#### 阶段4: 回合结算 (ROUND_END)
- 显示正确答案
- 显示本回合得分
- 清空画布
- **持续时间**: 5秒

#### 阶段5: 游戏结束 (GAME_END)
- 显示最终排名
- 返回房间或解散

### 5.3 计分规则

```typescript
// 猜对者得分
const calculateGuesserScore = (timeLeft: number, guesserCount: number): number => {
  const baseScore = 100
  const timeBonus = Math.floor(timeLeft * 2)  // 时间奖励
  const orderBonus = Math.max(0, 50 - guesserCount * 10)  // 顺序奖励
  return baseScore + timeBonus + orderBonus
}

// 画手得分
const calculateDrawerScore = (correctCount: number, totalPlayers: number): number => {
  const ratio = correctCount / (totalPlayers - 1)  // 排除自己
  return Math.floor(50 * ratio * correctCount)
}
```

### 6.2 提示生成算法

```typescript
const generateHints = (word: string): string[] => {
  const hints: string[] = []
  const chars = word.split('')
  
  // 显示第一个字
  hints.push(chars[0])
  
  // 中间用下划线表示
  for (let i = 1; i < chars.length - 1; i++) {
    hints.push('_')
  }
  
  // 如果超过2个字，显示最后一个字
  if (chars.length > 2) {
    hints.push(chars[chars.length - 1])
  } else if (chars.length === 2) {
    hints.push('_')
  }
  
  return hints
}

// 示例: "大象" -> ["大", "_"]
// 示例: "长颈鹿" -> ["长", "_", "鹿"]
// 示例: "冰淇淋" -> ["冰", "_", "淋"]
```

```typescript
  isDrawing: boolean
  hasGuessed: boolean
  socketId: string
}

// 房间
export interface RoomData {
  id: string
  name: string
  hostId: string
  players: PlayerData[]
  maxPlayers: number
  status: 'waiting' | 'playing' | 'finished'
  currentRound: number
  maxRounds: number
  currentDrawerId: string | null
  currentWord: string | null
  wordHints: string[]
  drawTime: number
}

// 游戏状态
export interface GameStateData {
  roomId: string
  status: 'waiting' | 'selecting' | 'drawing' | 'roundEnd' | 'gameEnd'
  currentWord: string | null
  wordHints: string[]
  timeLeft: number
  drawerId: string
  correctGuessers: string[]
  canvasData: DrawingElement[]
}

// 聊天消息
export interface ChatMessageData {
  id: string
  playerId: string
  playerName: string
  content: string
  type: 'chat' | 'system' | 'correct'
  timestamp: number
}

// Socket事件数据类型
export interface ClientToServerEvents {
  'room:create': (data: { name: string; maxPlayers: number }) => void
  'room:join': (data: { roomId: string; nickname: string }) => void
  'room:leave': (data: { roomId: string }) => void
  'room:ready': (data: { roomId: string }) => void
  'game:start': (data: { roomId: string }) => void

export interface ServerToClientEvents {
  'room:created': (data: { room: RoomData }) => void
  'room:joined': (data: { room: RoomData; player: PlayerData }) => void
  'room:updated': (data: { room: RoomData }) => void
  'room:error': (data: { message: string }) => void
  'game:started': (data: { gameState: GameStateData }) => void
  'game:selectWord': (data: { words: string[] }) => void
  'game:wordHint': (data: { hints: string[]; length: number }) => void
  'game:tick': (data: { timeLeft: number }) => void
  'game:correctGuess': (data: { playerId: string; playerName: string; score: number }) => void
  'game:roundEnd': (data: { word: string; scores: Record<string, number> }) => void
  'game:gameEnd': (data: { finalScores: Record<string, number> }) => void
  'canvas:sync': (data: { element: DrawingElement }) => void
  'canvas:cleared': () => void
  'chat:message': (data: { playerId: string; playerName: string; message: string; type: 'chat' | 'system' | 'correct' }) => void
```
